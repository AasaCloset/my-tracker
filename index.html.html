Google Sheets Integration Setup

Sync expenses and savings deposits directly into your personal Google Sheet using Google Apps Script!

3-Step Setup Instructions:

  1. Create a Google Sheet and click Extensions > Apps Script.
  2. Paste the code snippet below and click Deploy > New deployment.
  3. Select Web App, set Who has access to Anyone, deploy, and paste the Web App URL below.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Date", "Type", "Account", "Category/Goal", "Amount", "Currency", "Notes", "Timestamp"]);
  }
  
  sheet.appendRow([
    data.id,
    data.date,
    data.type,
    data.account,
    data.categoryOrGoal,
    data.amount,
    data.currency,
    data.notes,
    new Date()
  ]);
  
  return ContentService
    .createTextOutput(JSON.stringify({ "result": "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
